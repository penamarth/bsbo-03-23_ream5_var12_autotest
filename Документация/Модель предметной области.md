# Диаграмма классов

```
@startuml
skinparam classAttributeIconSize 0
hide empty members

' === Classes ===
class User {
    -userId : String
    -name : String
    -role : String
}

class Developer {
    +createTest() : TestCase
    +runTestSuite() : TestSession
}
Developer --|> User

class TestCase {
    -testId : String
    -name : String
    -scriptPath : String
    -expectedResult : String
    +execute() : TestResult
}

class TestSuite {
    -suiteId : String
    -name : String
    -tests : List<TestCase>
    +addTest(test: TestCase) : void
    +runAll() : List<TestResult>
}

class TestSession {
    -sessionId : String
    -startTime : DateTime
    -endTime : DateTime
    -status : String
    +generateReport() : TestReport
}

class Environment {
    -envId : String
    -type : String
    -config : Map<String, String>
    +isAvailable() : Boolean
}

class Resource {
    -resourceId : String
    -type : String
    -status : String
    +allocate() : Boolean
    +release() : void
}

class TestReport {
    -reportId : String
    -content : String
    -format : String
    +export(format: String) : void
}

class TestController {
    +startTestSession(suite: TestSuite, env: Environment) : TestSession
    +stopTestSession(session: TestSession) : void
}

class TestRunner {
    +runTest(test: TestCase, env: Environment) : TestResult
    +runSuite(suite: TestSuite, env: Environment) : List<TestResult>
}

class EnvironmentManager {
    +setupEnvironment(config: Map<String, String>) : Environment
    +teardownEnvironment(env: Environment) : void
}

class ResourceManager {
    +requestResource(type: String) : Resource
    +releaseResource(res: Resource) : void
}

class ReportGenerator {
    +generateReport(session: TestSession) : TestReport
    +sendNotification(report: TestReport) : void
}

' === Associations ===
Developer "1" --> "0..*" TestCase : creates
TestSuite "1" --> "1..*" TestCase : contains

TestController --> TestRunner : uses
TestController --> EnvironmentManager : uses
TestController --> ResourceManager : uses
TestController --> ReportGenerator : uses

TestSession --> TestSuite : runs
TestSession --> Environment : uses
TestSession --> TestReport : produces

EnvironmentManager --> Environment
ResourceManager --> Resource

TestRunner --> TestCase
TestRunner --> TestResult

@enduml
```
