' === Classes ===
class User {
    -userId : String
    -name : String
    -role : String
}

class Developer {
    +createTest() : TestCase
    +startTestSession(suite: TestSuite) : TestSession
    +stopTestSession(session: TestSession) : void
}
Developer --|> User

class TestController {
    +startSession(suite: TestSuite) : TestSession
    +stopSession(session: TestSession) : void
}

class TestCase {
    -testId : String
    -name : String
    -scriptPath : String
    -expectedResult : String
}

class TestSuite {
    -suiteId : String
    -name : String
    -tests : List<TestCase>
    +addTest(test: TestCase) : void
}

class TestSession {
    -sessionId : String
    -status : String
    -startTime : DateTime
    -endTime : DateTime
    +run() : List<TestResult>
    +finish() : TestReport
}

class TestResult {
    -resultId : String
    -status : String
    -actualResult : String
}

class Environment {
    -envId : String
    -type : String
    -config : Map<String, String>
    -isReady : Boolean
}

class EnvironmentManager {
    +setupEnvironment() : void
    +teardownEnvironment() : void
}

class TestRunner {
    +runSuite(
        suite: TestSuite,
        envManager: EnvironmentManager
    ) : List<TestResult>
}

class TestReport {
    -reportId : String
    -content : String
    -format : String
}

' === Relationships ===
Developer --> TestController : uses

TestController --> TestSession : creates
TestController --> TestRunner
TestController --> EnvironmentManager

TestSession --> TestSuite
TestSession --> TestReport : produces

TestRunner --> TestCase
TestRunner --> TestResult
TestRunner --> EnvironmentManager : uses

EnvironmentManager --> Environment : creates/configures

@enduml
