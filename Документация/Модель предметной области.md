@startuml
skinparam classAttributeIconSize 0
hide empty members

!define BOUNDED_CONTEXT(name) rectangle name <<Ограниченный контекст>>
!define AGGREGATE(name) rectangle name <<Агрегат>>

BOUNDED_CONTEXT("Контекст_Управление_Тестами") {

    class User {
        -userId : String
        -name : String
        -role : String
    }

    class Developer {
        +createTest() : TestCase
    }
    Developer --|> User

    AGGREGATE("Агрегат_TestSuite") {
        class TestSuite <<Корень агрегата>> {
            -suiteId : String
            -name : String
            -tests : List<TestCase>
            +addTest(test: TestCase)
        }

        class TestCase <<Сущность>> {
            -testId : String
            -name : String
            -scriptPath : String
            -expectedResult : String
        }

        TestSuite *-- TestCase
    }
}

BOUNDED_CONTEXT("Контекст_Выполнение_Тестов") {

    AGGREGATE("Агрегат_TestSession") {
        class TestSession <<Корень агрегата>> {
            -sessionId : String
            -status : String
            -startTime : DateTime
            -endTime : DateTime
            +run()
            +finish()
        }

        class TestResult <<Сущность>> {
            -resultId : String
            -status : String
            -actualResult : String
        }

        class TestReport <<Сущность>> {
            -reportId : String
            -content : String
            -format : String
        }

        TestSession *-- TestResult
        TestSession --> TestReport
    }

    class TestController <<Domain Service>> {
        +startSession(suite: TestSuite)
        +stopSession(session: TestSession)
    }

    class TestRunner <<Domain Service>> {
        +runSuite(suite: TestSuite)
    }
}

BOUNDED_CONTEXT("Контекст_Окружение") {

    class Environment <<Сущность>> {
        -envId : String
        -type : String
        -config : Map<String, String>
        -isReady : Boolean
    }

    class EnvironmentManager <<Domain Service>> {
        +setupEnvironment()
        +teardownEnvironment()
    }

    EnvironmentManager --> Environment
}

TestController --> TestSuite
TestController --> TestSession
TestRunner --> EnvironmentManager

@enduml
